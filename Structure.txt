'use client';
import { useEffect, useState } from 'react';
import { useParams } from 'next/navigation';
import Link from 'next/link';
import { api } from '@/lib/api';
import { formatPrice, formatDate } from '@/lib/format';

const STATUS_LABELS = {
  pending: 'در انتظار پرداخت',
  processing: 'در حال پردازش',
  shipped: 'ارسال شده',
  delivered: 'تحویل شده',
  cancelled: 'لغو شده',
};

export default function MyOrderDetailPage() {
  const { id } = useParams();
  const [order, setOrder] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.getMyOrder(id)
      .then(data => setOrder(data.order))
      .catch(console.error)
      .finally(() => setLoading(false));
  }, [id]);

  if (loading) return <div className="text-ash animate-pulse">در حال بارگذاری...</div>;
  if (!order) return <div className="text-ember">سفارش یافت نشد</div>;

  return (
    <div className="space-y-6">
      <div className="flex items-center gap-3">
        <Link href="/account/orders" className="text-ash hover:text-ink transition">→</Link>
        <h1 className="text-lg font-bold text-ink font-mono" dir="ltr">{order.code}</h1>
      </div>

      {/* مسیر وضعیت سفارش */}
      {order.statusHistory?.length > 0 && (
        <div className="bg-panel border border-line rounded-card p-5">
          <h2 className="font-bold text-ink text-sm mb-4">وضعیت سفارش</h2>
          <div className="space-y-3">
            {order.statusHistory.map((h, i) => (
              <div key={i} className="flex items-center gap-3 text-sm">
                <span className="w-2 h-2 rounded-full bg-grape shrink-0" />
                <span className="text-ink">{STATUS_LABELS[h.status] || h.status}</span>
                <span className="text-ash-dark text-xs mr-auto">{formatDate(h.date || h.createdAt)}</span>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* اقلام سفارش */}
      <div className="bg-panel border border-line rounded-card p-5">
        <h2 className="font-bold text-ink text-sm mb-4">اقلام سفارش</h2>
        <div className="space-y-3">
          {order.items?.map((item, i) => (
            <div key={i} className="flex justify-between text-sm">
              <span className="text-ash">{item.name || item.product?.name} × {item.quantity?.toLocaleString('fa-IR')}</span>
              <span className="text-ink font-mono">{formatPrice((item.price || 0) * item.quantity)}</span>
            </div>
          ))}
        </div>
        <div className="border-t border-line mt-4 pt-4 space-y-1.5">
          <Row label="جمع کالاها" value={formatPrice(order.subtotal)} />
          <Row label="هزینه ارسال" value={formatPrice(order.shippingCost)} />
          {order.discount > 0 && <Row label="تخفیف" value={`- ${formatPrice(order.discount)}`} />}
          <Row label="مبلغ نهایی" value={formatPrice(order.total)} bold />
        </div>
      </div>

      {/* آدرس گیرنده */}
      {order.shippingInfo && (
        <div className="bg-panel border border-line rounded-card p-5">
          <h2 className="font-bold text-ink text-sm mb-3">آدرس گیرنده</h2>
          <p className="text-sm text-ash leading-7">
            {order.shippingInfo.name} — <span dir="ltr">{order.shippingInfo.phone}</span><br />
            {order.shippingInfo.city}، {order.shippingInfo.address}<br />
            کد پستی: <span dir="ltr">{order.shippingInfo.postalCode}</span>
          </p>
        </div>
      )}
    </div>
  );
}

function Row({ label, value, bold }) {
  return (
    <div className={`flex justify-between text-sm ${bold ? 'text-ink font-bold pt-2 border-t border-line' : 'text-ash'}`}>
      <span>{label}</span>
      <span className="font-mono">{value}</span>
    </div>
  );
}

'use client';
import { useEffect } from 'react';

// نگهبان بخش حساب کاربری — اگه لاگین نبود میفرسته به /login
export default function AccountLayout({ children }) {
  const { user, loading, logout } = useAuth();
  const router = useRouter();
  const pathname = usePathname();

  useEffect(() => {
    if (!loading && !user) router.push('/login');
  }, [loading, user, router]);

  if (loading || !user) return <div className="text-center py-24 text-ash">در حال بارگذاری...</div>;

  const links = [
    { href: '/account', label: 'پروفایل من', icon: '👤' },
    { href: '/account/orders', label: 'سفارش‌های من', icon: '📦' },
  ];

  return (
    <div className="max-w-5xl mx-auto px-4 md:px-6 py-8">
      <div className="flex flex-col md:flex-row gap-6">
        <aside className="md:w-56 shrink-0">
          <div className="bg-panel border border-line rounded-card p-3 space-y-1">
            {links.map(link => (
              <Link
                key={link.href}
                href={link.href}
                className={`flex items-center gap-2 px-3 py-2.5 rounded-btn text-sm transition ${
                  pathname === link.href ? 'bg-grape text-white' : 'text-ash hover:text-ink hover:bg-card'
                }`}
              >
                <span>{link.icon}</span> {link.label}
              </Link>
            ))}
            <button
              onClick={() => { logout(); router.push('/'); }}
              className="w-full flex items-center gap-2 px-3 py-2.5 rounded-btn text-sm text-ember hover:bg-ember/10 transition"
            >
              🚪 خروج از حساب
            </button>
          </div>

      </div>
    </div>
  );

import { notFound } from 'next/navigation';
import Link from 'next/link';
import { api } from '@/lib/api';
import { buildArticleJsonLd, buildBreadcrumbJsonLd } from '@/lib/seo';
import { formatDate } from '@/lib/format';
import Breadcrumbs from '@/components/common/Breadcrumbs';
import ArticleContent, { extractHeadings } from '@/components/blog/ArticleContent';
import ShareButtons from '@/components/blog/ShareButtons';
import BlogCard from '@/components/blog/BlogCard';

const SITE_URL = process.env.NEXT_PUBLIC_SITE_URL || 'https://technubela.example.com';
const API_BASE = process.env.NEXT_PUBLIC_API_BASE || 'http://localhost:5000';

// ============================================================
// صفحه‌ی مقاله — طبق خواسته‌ی شما، سئوش با دقت کامل رعایت شده:
// - متادیتای کامل + canonical + Open Graph نوع "article"
// - JSON-LD از نوع Article (نویسنده، تاریخ انتشار/ویرایش، تصویر)
// - JSON-LD Breadcrumb
// - ساختار semantic (h1 برای عنوان، h2 برای زیرعنوان‌ها)
// - فهرست مطالب لینک‌دار (کمک به دسترسی‌پذیری و dwell time)
// - لینک‌دهی داخلی به مقالات مرتبط و محصولات دسته‌ی مرتبط
// ============================================================

async function getPostSafe(slug) {
  try {
    const data = await api.getBlogPost(slug);
    return data.post;
  } catch {
    return null;
  }
}

export async function generateMetadata({ params }) {
  const { slug } = await params;
  const post = await getPostSafe(slug);
  if (!post) return {};

  const image = post.image ? `${SITE_URL}${post.image}` : undefined;

  return {
    title: post.title,
    description: post.excerpt,
    alternates: { canonical: `/blog/${post.slug}` },
    openGraph: {
      type: 'article',
      title: post.title,
      description: post.excerpt,
      images: image ? [image] : [],
      publishedTime: post.date,
      modifiedTime: post.updatedDate || post.date,
      authors: [post.author?.name].filter(Boolean),
    },
    twitter: { card: 'summary_large_image', title: post.title, description: post.excerpt },
  };
}

export default async function BlogPostPage({ params }) {
  const { slug } = await params;
  const post = await getPostSafe(slug);
  if (!post) notFound();

  const relatedData = await api.getBlogPosts(`?category=${encodeURIComponent(post.category)}&limit=4`).catch(() => ({ posts: [] }));
  const related = (relatedData.posts || []).filter(p => p.slug !== post.slug).slice(0, 3);

  const headings = extractHeadings(post.content);
  const articleJsonLd = buildArticleJsonLd(post, SITE_URL);
  const breadcrumbItems = [
    { name: 'خانه', href: '/' },
    { name: 'مجله', href: '/blog' },
    { name: post.title, href: `/blog/${post.slug}` },
  ];
  const breadcrumbJsonLd = buildBreadcrumbJsonLd(breadcrumbItems, SITE_URL);

  return (
    <div className="max-w-6xl mx-auto px-4 md:px-6 py-8">
      {/* eslint-disable-next-line react/no-danger */}
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(articleJsonLd) }} />
      {/* eslint-disable-next-line react/no-danger */}
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(breadcrumbJsonLd) }} />

      <div className="mb-4">
        <Breadcrumbs items={breadcrumbItems} />
      </div>

      <div className="flex flex-col lg:flex-row gap-8">
        <article className="flex-1 min-w-0 max-w-3xl">
          <span className="text-xs text-lagoon">{post.category}</span>
          <h1 className="text-2xl md:text-3xl font-bold text-ink mt-2 mb-4 leading-10">{post.title}</h1>

          <div className="flex items-center gap-3 mb-6">
            <div className="w-9 h-9 rounded-full bg-grape flex items-center justify-center text-white text-sm font-bold shrink-0">
              {post.author?.avatar || 'ت'}
            </div>
            <div>
              <p className="text-sm text-ink">{post.author?.name}</p>
              <p className="text-xs text-ash-dark">{post.author?.role} · {formatDate(post.date)} · {post.readTime?.toLocaleString('fa-IR')} دقیقه مطالعه</p>
            </div>
          </div>

          {post.image && (
            <img src={`${API_BASE}${post.image}`} alt={post.title} className="w-full aspect-video object-cover rounded-card mb-6" />
          )}

          <p className="text-ash text-base leading-8 mb-6 border-r-2 border-grape pr-4">{post.excerpt}</p>

          <ArticleContent blocks={post.content} />

          {post.tags?.length > 0 && (
            <div className="flex flex-wrap gap-2 mt-8 pt-6 border-t border-line">
              {post.tags.map((tag , i) => (
                <span key={i} className="text-xs bg-card text-ash px-3 py-1.5 rounded-full">#{tag}</span>
              ))}
            </div>
          )}

          <div className="mt-6 pt-6 border-t border-line">
            <ShareButtons url={`${SITE_URL}/blog/${post.slug}`} title={post.title} />
          </div>
        </article>

        {/* فهرست مطالب — فقط دسکتاپ */}
        {headings.length > 0 && (
          <aside className="hidden lg:block w-56 shrink-0">
            <div className="sticky top-24 bg-panel border border-line rounded-card p-4">
              <h2 className="text-xs font-bold text-ink mb-3">فهرست مطالب</h2>
              <ul className="space-y-2">
                {headings.map(h => (
                  <li key={h.id}>
                    <a href={`#${h.id}`} className="text-xs text-ash hover:text-ink transition line-clamp-2">{h.text}</a>
                  </li>
                ))}
              </ul>
            </div>
          </aside>
        )}
      </div>

      {related.length > 0 && (
        <section className="mt-12 max-w-3xl">
          <h2 className="text-lg font-bold text-ink mb-4">مقالات مرتبط</h2>
          <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
            {related.map(p => <BlogCard key={p.id} post={p} />)}
          </div>
        </section>
      )}
    </div>
  );
}

import Link from 'next/link';
import { api } from '@/lib/api';
import BlogCard from '@/components/blog/BlogCard';
import Pagination from '@/components/common/Pagination';
import Breadcrumbs from '@/components/common/Breadcrumbs';

// ============================================================
// صفحه‌ی وبلاگ — دومین اولویت سئوی این پروژه بعد از صفحه‌ی محصول.
// Server Component، متادیتای کامل، و لینک‌دهی داخلی به مقالات.
// ============================================================

export const metadata = {
  title: 'مجله تک‌نبولا | آموزش، مقایسه و راهنمای خرید قطعات کامپیوتر',
  description: 'جدیدترین مقالات آموزشی، مقایسه‌ی محصولات، و راهنمای خرید قطعات کامپیوتر و گجت از تیم تحریریه‌ی تک‌نبولا.',
  alternates: { canonical: '/blog' },
};

export default async function BlogPage({ searchParams }) {
  const sp = await searchParams;
  const page = Number(sp.page) || 1;
  const params = new URLSearchParams({ page: String(page), limit: '9' });
  if (sp.category) params.set('category', sp.category);

  const data = await api.getBlogPosts(`?${params.toString()}`).catch(() => ({ posts: [], total: 0, totalPages: 1 }));
  const posts = data.posts || [];
  const totalPages = data.totalPages || 1;

  // لیست دسته‌بندی‌های مقالات از خود پست‌های همین صفحه استخراج میشه
  // (چون بک‌اند مدل جدای category برای بلاگ نداره، دقیقاً مثل محصولات)
  const categories = [...new Set(posts.map(p => p.category))];

  return (
    <div className="max-w-6xl mx-auto px-4 md:px-6 py-8">
      <div className="mb-4">
        <Breadcrumbs items={[{ name: 'خانه', href: '/' }, { name: 'مجله', href: '/blog' }]} />
      </div>

      <h1 className="text-2xl font-bold text-ink mb-2">مجله تک‌نبولا</h1>
      <p className="text-ash text-sm mb-6">آموزش، مقایسه و راهنمای خرید — نوشته‌شده توسط کارشناسان سخت‌افزار</p>

      {categories.length > 0 && (
        <div className="flex gap-2 flex-wrap mb-6">
          <Link href="/blog" className={`text-xs px-3 py-1.5 rounded-full transition ${!sp.category ? 'bg-grape text-white' : 'bg-panel text-ash hover:text-ink'}`}>
            همه
          </Link>
          {categories.map(cat => (
            <Link
              key={cat}
              href={`/blog?category=${encodeURIComponent(cat)}`}
              className={`text-xs px-3 py-1.5 rounded-full transition ${sp.category === cat ? 'bg-grape text-white' : 'bg-panel text-ash hover:text-ink'}`}
            >
              {cat}
            </Link>
          ))}
        </div>
      )}

      {posts.length === 0 ? (
        <p className="text-ash text-sm py-10 text-center">مقاله‌ای پیدا نشد.</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-3 gap-5">
          {posts.map(post => <BlogCard key={post.id} post={post} />)}
        </div>
      )}

      <Pagination page={page} totalPages={totalPages} />
    </div>
  );
}

export async function generateStaticParams() {
  return ALL_CATEGORIES.map(c => ({ slug: c.slug }));
}

export async function generateMetadata({ params }) {
  const { slug } = await params;
  const category = ALL_CATEGORIES.find(c => c.slug === slug);
  if (!category) return {};

  return {
    title: `خرید ${category.label} | قیمت روز و مشخصات فنی`,
    description: `بهترین قیمت خرید ${category.label} با گارانتی اصالت و ارسال سریع. مقایسه و فیلتر بر اساس برند و قیمت در فروشگاه تک‌نبولا.`,
    alternates: { canonical: `/category/${slug}` },
  };
}

export default async function CategoryPage({ params, searchParams }) {
  const { slug } = await params;
  const sp = await searchParams;

  const category = ALL_CATEGORIES.find(c => c.slug === slug);
  if (!category) notFound();

  const query = buildProductQuery(sp, { category: slug });
  const [productsData, facetsData] = await Promise.allSettled([
    api.getProducts(query),
    api.getProductFacets(`?category=${slug}`),
  ]);

  const products = productsData.status === 'fulfilled' ? (productsData.value.products || []) : [];
  const total = productsData.status === 'fulfilled' ? (productsData.value.total ?? products.length) : 0;
  const totalPages = productsData.status === 'fulfilled' ? (productsData.value.totalPages ?? 1) : 1;
  const facets = facetsData.status === 'fulfilled' ? facetsData.value : { brands: [] };
  const page = Number(sp.page) || 1;

  return (
    <div className="max-w-7xl mx-auto px-4 md:px-6 py-6">
      <div className="mb-4">
        <Breadcrumbs items={[{ name: 'خانه', href: '/' }, { name: category.label, href: `/category/${slug}` }]} />
      </div>

      <div className="flex items-center gap-3 mb-6">
        <span className="text-3xl">{category.icon}</span>
        <h1 className="text-xl md:text-2xl font-bold text-ink">{category.label}</h1>
      </div>

      <div className="flex flex-col lg:flex-row gap-6">
        <div className="hidden lg:block">
          <Suspense fallback={null}>
            <FilterSidebar facets={facets} lockedCategory={slug} />
          </Suspense>
        </div>

        <div className="flex-1 min-w-0">
          <Suspense fallback={null}>
            <ProductToolbar total={total} facets={facets} lockedCategory={slug} />
            <ActiveFilters lockedCategory={slug} />
          </Suspense>

          {products.length === 0 ? (
            <div className="text-center py-20 bg-panel border border-line rounded-card">
              <div className="text-5xl mb-4">{category.icon}</div>
              <p className="text-ink font-bold mb-1">فعلاً محصولی توی این دسته‌بندی نیست</p>
              <p className="text-sm text-ash">به‌زودی محصولات جدید اضافه میشه.</p>
            </div>
          ) : (
            <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-4 gap-4">
              {products.map(p => <ProductCard key={p.id} product={p} />)}
            </div>
          )}

          <Suspense fallback={null}>
            <Pagination page={page} totalPages={totalPages} />
          </Suspense>
        </div>
      </div>
    </div>
  );
}

// ============================================================
// صفحه‌ی بازگشت از درگاه پرداخت
// ------------------------------------------------------------
// زرین‌پال کاربر رو به این آدرس با کوئری‌پارام‌های Authority و
// Status برمی‌گردونه (دقیقاً همین اسم‌ها، با حرف بزرگ — استاندارد
// خود زرین‌پاله). orderId رو هم خودمون موقع ساخت لینک پرداخت
// (callback_url) بهش اضافه کرده بودیم تا بفهمیم این پرداخت مال
// کدوم سفارشه.
// ============================================================

import { useEffect, useState, Suspense } from 'react';
import { useSearchParams } from 'next/navigation';
import Link from 'next/link';
import { api } from '@/lib/api';
import { formatPrice } from '@/lib/format';

function CallbackContent() {
  const searchParams = useSearchParams();
  const [status, setStatus] = useState('loading'); // loading | success | failed
  const [order, setOrder] = useState(null);
  const [error, setError] = useState('');

  useEffect(() => {
    const orderId = searchParams.get('orderId');
    const authority = searchParams.get('Authority');
    const payStatus = searchParams.get('Status'); // 'OK' یا 'NOK'

    if (!orderId || !authority) {
      setStatus('failed');
      setError('اطلاعات پرداخت ناقصه.');
      return;
    }

    api.verifyPayment(orderId, authority, payStatus)
      .then(data => { setOrder(data.order); setStatus('success'); })
      .catch(err => { setError(err.message); setStatus('failed'); });
  }, [searchParams]);

  if (status === 'loading') {
    return (
      <div className="text-center py-24">
        <div className="animate-pulse text-4xl mb-4">⏳</div>
        <p className="text-ash">در حال بررسی نتیجه‌ی پرداخت...</p>
      </div>
    );
  }

  if (status === 'failed') {
    return (
      <div className="text-center py-24 max-w-md mx-auto">
        <div className="text-5xl mb-4">❌</div>
        <h1 className="text-ink font-bold text-lg mb-2">پرداخت ناموفق بود</h1>
        <p className="text-ash text-sm mb-6">{error || 'مبلغ کسر نشد. می‌تونید دوباره تلاش کنید.'}</p>
        <Link href="/cart" className="bg-grape hover:bg-grape/80 text-white font-bold px-6 py-3 rounded-btn transition inline-block">
          بازگشت به سبد خرید
        </Link>
      </div>
    );
  }

  return (
    <div className="text-center py-24 max-w-md mx-auto">
      <div className="text-5xl mb-4">✅</div>
      <h1 className="text-ink font-bold text-lg mb-2">پرداخت با موفقیت انجام شد</h1>
      {order && (
        <p className="text-ash text-sm mb-1">
          کد سفارش: <span className="font-mono text-ink" dir="ltr">{order.code}</span>
        </p>
      )}
      {order?.total && <p className="text-ash text-sm mb-6">مبلغ پرداخت‌شده: {formatPrice(order.total)}</p>}
      <div className="flex items-center justify-center gap-3">
        <Link href="/account/orders" className="bg-grape hover:bg-grape/80 text-white font-bold px-5 py-2.5 rounded-btn transition">
          پیگیری سفارش
        </Link>
        <Link href="/products" className="bg-panel border border-line text-ink px-5 py-2.5 rounded-btn transition">
          ادامه خرید
        </Link>
      </div>
    </div>
  );
}

'use client';
// ============================================================
// صفحه‌ی ورود — کد یک‌بارمصرف پیامکی
// ------------------------------------------------------------
// این صفحه فقط از endpoint های موجود بک‌اند استفاده می‌کنه
// (/auth/send-otp و /auth/verify-otp، همون‌هایی که پنل ادمین هم
// استفاده می‌کنه) — پس با وصل شدن SMS.ir به بک‌اند، اینجا هیچ
// تغییری لازم نداره؛ فقط پیامک واقعی به‌جای هر مکانیزم تستی فعلی
// ارسال میشه.
// ============================================================

import { useState, useRef } from 'react';
import { useRouter } from 'next/navigation';
import { api } from '@/lib/api';
import { useAuth } from '@/contexts/AuthContext';

export default function LoginPage() {
  const router = useRouter();
  const { login } = useAuth();
  const [step, setStep] = useState('phone'); // 'phone' | 'otp'
  const [phone, setPhone] = useState('');
  const [code, setCode] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [resendTimer, setResendTimer] = useState(0);
  const timerRef = useRef(null);

  const startResendTimer = () => {
    setResendTimer(60);
    clearInterval(timerRef.current);
    timerRef.current = setInterval(() => {
      setResendTimer(t => {
        if (t <= 1) { clearInterval(timerRef.current); return 0; }
        return t - 1;
      });
    }, 1000);
  };

  const handleSendOTP = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);
    try {
      await api.sendOTP(phone);
      setStep('otp');
      startResendTimer();
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  const handleVerify = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);
    try {
      const data = await api.verifyOTP(phone, code);
      login(data.token, data.user);
      router.push('/account');
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-sm mx-auto px-4 py-16">
      <h1 className="text-xl font-bold text-ink text-center mb-2">ورود به تک‌نبولا</h1>
      <p className="text-sm text-ash text-center mb-8">
        {step === 'phone' ? 'شماره موبایل خودتون رو وارد کنید' : `کد ارسال‌شده به ${phone} رو وارد کنید`}
      </p>

      {error && <div className="bg-ember/10 text-ember text-sm px-4 py-3 rounded-btn mb-4">{error}</div>}

      {step === 'phone' ? (
        <form onSubmit={handleSendOTP} className="space-y-4">
          <input
            type="tel"
            dir="ltr"
            required
            value={phone}
            onChange={e => setPhone(e.target.value)}
            placeholder="09123456789"
            pattern="09[0-9]{9}"
            className="w-full bg-panel border border-line rounded-btn px-4 py-3 text-ink text-center tracking-wider focus:border-grape transition"
          />
          <button
            type="submit"
            disabled={loading}
            className="w-full bg-grape hover:bg-grape/80 text-white font-bold py-3 rounded-btn transition disabled:opacity-50"
          >
            {loading ? 'در حال ارسال...' : 'دریافت کد ورود'}
          </button>
        </form>
      ) : (
        <form onSubmit={handleVerify} className="space-y-4">
          <input
            type="text"
            dir="ltr"
            required
            inputMode="numeric"
            maxLength={6}
            value={code}
            onChange={e => setCode(e.target.value.replace(/\D/g, ''))}
            placeholder="کد ۴ رقمی"
            className="w-full bg-panel border border-line rounded-btn px-4 py-3 text-ink text-center text-xl tracking-[0.5em] focus:border-grape transition"
          />
          <button
            type="submit"
            disabled={loading}
            className="w-full bg-grape hover:bg-grape/80 text-white font-bold py-3 rounded-btn transition disabled:opacity-50"
          >
            {loading ? 'در حال بررسی...' : 'تایید و ورود'}
          </button>
          <div className="text-center">
            {resendTimer > 0 ? (
              <span className="text-xs text-ash-dark">ارسال مجدد کد تا {resendTimer.toLocaleString('fa-IR')} ثانیه دیگر</span>
            ) : (
              <button type="button" onClick={handleSendOTP} className="text-xs text-grape-light hover:underline">
                ارسال مجدد کد
              </button>
            )}
          </div>
          <button type="button" onClick={() => setStep('phone')} className="w-full text-xs text-ash hover:text-ink transition">
            ← ویرایش شماره موبایل
          </button>
        </form>
      )}
    </div>
  );
}

// صفحه‌ی جزئیات محصول
// ------------------------------------------------------------
// این صفحه از نظر سئو مهم‌ترین صفحه‌ی فروشگاهه: متادیتای کامل،
// JSON-LD محصول (قیمت/موجودی/امتیاز برای Rich Results گوگل)،
// و JSON-LD Breadcrumb. همه‌چیز سمت سرور رندر میشه.
// ============================================================

async function getProductSafe(param) {
  try {
    const id = extractProductId(param);
    const data = await api.getProduct(id);
    return data.product;
  } catch {
    return null;
  }
}

export async function generateMetadata({ params }) {
  const { id } = await params;
  const product = await getProductSafe(id);
  if (!product) return {};

  const description = product.desc?.slice(0, 155) || `خرید ${product.name} با بهترین قیمت و گارانتی اصالت`;
  const image = product.images?.[0] ? `${SITE_URL}${product.images[0]}` : undefined;

  return {
    title: `${product.name} | قیمت و خرید`,
    description,
    alternates: { canonical: productUrl(product) },
    openGraph: {
      title: product.name,
      description,
      images: image ? [image] : [],
      type: 'website',
    },
  };
}

  const [reviewsData, relatedData] = await Promise.allSettled([

  const related = relatedData.status === 'fulfilled'

export default async function ProductsPage({ searchParams }) {
  const sp = await searchParams;
  const query = buildProductQuery(sp);

  // fetch محصولات و facet ها (برای گزینه‌های فیلتر) به‌صورت موازی
  const [productsData, facetsData] = await Promise.allSettled([
    api.getProducts(query),
    api.getProductFacets(sp.category ? `?category=${sp.category}` : ''),
  ]);

  const products = productsData.status === 'fulfilled' ? (productsData.value.products || []) : [];
  const total = productsData.status === 'fulfilled' ? (productsData.value.total ?? products.length) : 0;
  const totalPages = productsData.status === 'fulfilled' ? (productsData.value.totalPages ?? 1) : 1;
  const facets = facetsData.status === 'fulfilled' ? facetsData.value : { brands: [] };
  const page = Number(sp.page) || 1;

  return (
    <div className="max-w-7xl mx-auto px-4 md:px-6 py-6">
      <div className="mb-4">
        <Breadcrumbs items={[{ name: 'خانه', href: '/' }, { name: 'همه محصولات', href: '/products' }]} />
      </div>

      <h1 className="text-xl md:text-2xl font-bold text-ink mb-6">
        {sp.search ? `نتایج جستجو برای «${sp.search}»` : 'همه محصولات'}
      </h1>

      <div className="flex flex-col lg:flex-row gap-6">
        <div className="hidden lg:block">
          <Suspense fallback={null}>
            <FilterSidebar facets={facets} />
          </Suspense>
        </div>

        <div className="flex-1 min-w-0">
          <Suspense fallback={null}>
            <ProductToolbar total={total} facets={facets} />
            <ActiveFilters />
          </Suspense>

          {products.length === 0 ? (
            <EmptyState />
          ) : (
            <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-4 gap-4">
              {products.map(p => <ProductCard key={p.id} product={p} />)}
            </div>
          )}

          <Suspense fallback={null}>
            <Pagination page={page} totalPages={totalPages} />
          </Suspense>
        </div>
      </div>
    </div>
  );
}

import { api } from '@/lib/api';
import { ALL_CATEGORIES } from '@/lib/categories';
import { productUrl } from '@/lib/seo';

const SITE_URL = process.env.NEXT_PUBLIC_SITE_URL || 'https://technubela.example.com';

// ============================================================
// sitemap.xml خودکار — Next.js از روی این فایل، آدرس
// /sitemap.xml رو خودش می‌سازه (نیازی به فایل استاتیک نیست).
// ⚠️ اگه تعداد محصولات/مقالات خیلی زیاد بشه (چند هزارتا)، بهتره
// این فایل رو به چند sitemap تقسیم کنی (generateSitemaps) —
// برای الان که کاتالوگ در حد یه فروشگاه معمولیه، یه فایل کافیه.
// ============================================================

export default async function sitemap() {
  const staticPages = [
    { url: `${SITE_URL}/`, changeFrequency: 'daily', priority: 1 },
    { url: `${SITE_URL}/products`, changeFrequency: 'daily', priority: 0.9 },
    { url: `${SITE_URL}/blog`, changeFrequency: 'daily', priority: 0.8 },
  ];

  const categoryPages = ALL_CATEGORIES.map(c => ({
    url: `${SITE_URL}/category/${c.slug}`,
    changeFrequency: 'daily',
    priority: 0.7,
  }));

  let productPages = [];
  let blogPages = [];

  try {
    const productsData = await api.getProducts('?limit=1000');
    productPages = (productsData.products || []).map(p => ({
      url: `${SITE_URL}${productUrl(p)}`,
      lastModified: p.updatedAt,
      changeFrequency: 'weekly',
      priority: 0.6,
    }));
  } catch {
    // اگه بک‌اند در دسترس نبود، فقط صفحات ثابت توی سایت‌مپ می‌مونن
  }

  try {
    const blogData = await api.getBlogPosts('?limit=1000');
    blogPages = (blogData.posts || []).map(p => ({
      url: `${SITE_URL}/blog/${p.slug}`,
      lastModified: p.updatedDate || p.date,
      changeFrequency: 'monthly',
      priority: 0.5,
    }));
  } catch {
    // همون‌طور
  }

  return [...staticPages, ...categoryPages, ...productPages, ...blogPages];
}

'use client';
import { useRouter, useSearchParams, usePathname } from 'next/navigation';

// صفحه‌بندی — با آپدیت پارامتر page توی URL (نه state داخلی)، تا
// لینک هر صفحه قابل اشتراک‌گذاری/ایندکس‌شدن توسط گوگل باشه.
export default function Pagination({ page, totalPages }) {
  const router = useRouter();
  const pathname = usePathname();
  const searchParams = useSearchParams();

  if (totalPages <= 1) return null;

  const goTo = (p) => {
    const params = new URLSearchParams(searchParams.toString());
    params.set('page', p);
    router.push(`${pathname}?${params.toString()}`, { scroll: true });
  };

  // نمایش حداکثر ۵ شماره صفحه با نقطه‌چین برای بقیه
  const pages = [];
  const start = Math.max(1, page - 2);
  const end = Math.min(totalPages, start + 4);
  for (let p = start; p <= end; p++) pages.push(p);

  return (
    <div className="flex items-center justify-center gap-2 mt-8">
      <button
        onClick={() => goTo(page - 1)}
        disabled={page <= 1}
        className="w-9 h-9 rounded-btn bg-panel text-ash hover:text-ink disabled:opacity-30 transition"
        aria-label="صفحه قبل"
      >
        ›
      </button>
      {start > 1 && <span className="text-ash-dark px-1">…</span>}
      {pages.map(p => (
        <button
          key={p}
          onClick={() => goTo(p)}
          className={`w-9 h-9 rounded-btn text-sm transition ${
            p === page ? 'bg-grape text-white font-bold' : 'bg-panel text-ash hover:text-ink'
          }`}
        >
          {p.toLocaleString('fa-IR')}
        </button>
      ))}
      {end < totalPages && <span className="text-ash-dark px-1">…</span>}
      <button
        onClick={() => goTo(page + 1)}
        disabled={page >= totalPages}
        className="w-9 h-9 rounded-btn bg-panel text-ash hover:text-ink disabled:opacity-30 transition"
        aria-label="صفحه بعد"
      >
        ‹
      </button>
    </div>
  );
}

'use client';
import { useRouter, usePathname, useSearchParams } from 'next/navigation';
import { categoryLabel } from '@/lib/categories';
import { formatPrice } from '@/lib/format';

// ردیف چیپ‌های فیلتر فعال — شفافیت کامل به کاربر که الان دقیقاً
// چی فیلتر شده، با امکان حذف تکی هر فیلتر.
export default function ActiveFilters({ lockedCategory }) {
  const router = useRouter();
  const pathname = usePathname();
  const searchParams = useSearchParams();

  const chips = [];

  if (!lockedCategory) {
    (searchParams.get('category')?.split(',').filter(Boolean) || []).forEach(c =>
      chips.push({ key: 'category', value: c, label: categoryLabel(c) })
    );
  }
  (searchParams.get('brand')?.split(',').filter(Boolean) || []).forEach(b =>
    chips.push({ key: 'brand', value: b, label: b })
  );
  if (searchParams.get('minPrice') || searchParams.get('maxPrice')) {
    const min = searchParams.get('minPrice');
    const max = searchParams.get('maxPrice');
    chips.push({
      key: 'price', value: '__price__',
      label: `قیمت: ${min ? formatPrice(min) : '۰'} تا ${max ? formatPrice(max) : '∞'}`,
    });
  }
  if (searchParams.get('minRating')) chips.push({ key: 'minRating', value: '__rating__', label: `${searchParams.get('minRating')}★ به بالا` });
  if (searchParams.get('inStock') === 'true') chips.push({ key: 'inStock', value: '__stock__', label: 'فقط موجود' });
  if (searchParams.get('isNew') === 'true') chips.push({ key: 'isNew', value: '__new__', label: 'جدید' });
  if (searchParams.get('isBestseller') === 'true') chips.push({ key: 'isBestseller', value: '__best__', label: 'پرفروش' });
  if (searchParams.get('search')) chips.push({ key: 'search', value: '__search__', label: `جستجو: ${searchParams.get('search')}` });

  if (chips.length === 0) return null;

  function removeChip(chip) {
    const params = new URLSearchParams(searchParams.toString());
    if (chip.key === 'category' || chip.key === 'brand') {
      const list = (params.get(chip.key)?.split(',').filter(Boolean) || []).filter(v => v !== chip.value);
      if (list.length) params.set(chip.key, list.join(',')); else params.delete(chip.key);
    } else if (chip.key === 'price') {
      params.delete('minPrice'); params.delete('maxPrice');
    } else {
      params.delete(chip.key);
    }
    params.delete('page');
    router.push(`${pathname}?${params.toString()}`, { scroll: false });
  }

  return (
    <div className="flex flex-wrap gap-2 mb-4">
      {chips.map(chip => (
        <button
          key={`${chip.key}-${chip.value}`}
          onClick={() => removeChip(chip)}
          className="flex items-center gap-1.5 bg-card border border-line text-xs text-ink px-3 py-1.5 rounded-full hover:border-ember transition"
        >
          {chip.label} <span className="text-ash-dark">✕</span>
        </button>
      ))}
    </div>
  );
}

'use client';
import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { useCart } from '@/contexts/CartContext';
import { formatPrice, discountPercent } from '@/lib/format';

export default function AddToCartBox({ product }) {
  const router = useRouter();
  const { addItem } = useCart();
  const [qty, setQty] = useState(1);
  const [added, setAdded] = useState(false);
  const outOfStock = !product.stock || product.stock <= 0;
  const discount = discountPercent(product.price, product.oldPrice);

  const handleAdd = () => {
    addItem(product, qty);
    setAdded(true);
    setTimeout(() => setAdded(false), 2000);
  };

  const handleBuyNow = () => {
    addItem(product, qty);
    router.push('/checkout');
  };

  return (
    <div className="bg-panel border border-line rounded-card p-5 space-y-4">
      <div className="flex items-baseline gap-3 font-mono">
        {product.oldPrice > product.price && (
          <>
            <span className="text-ash-dark line-through text-sm">{formatPrice(product.oldPrice)}</span>
            <span className="bg-ember/15 text-ember text-xs font-bold px-2 py-1 rounded">٪{discount.toLocaleString('fa-IR')} تخفیف</span>
          </>
        )}
      </div>
      <div className="text-2xl font-bold text-ink font-mono">{formatPrice(product.price)}</div>

      {outOfStock ? (
        <p className="text-ember text-sm font-bold">این محصول در حال حاضر ناموجود است</p>

          <div className="flex items-center gap-3">

// ============================================================
// فیلتر محصولات — مهم‌ترین بخش UX این پروژه طبق خواسته‌ی کارفرما
// ------------------------------------------------------------
// طراحی:
// - همه‌ی فیلترها توی URL query string ذخیره میشن (نه state
//   داخلی تنها) → یعنی کاربر می‌تونه لینک نتیجه‌ی فیلترشده رو
//   کپی/شیر کنه، دکمه‌ی back مرورگر درست کار می‌کنه، و گوگل هم
//   می‌تونه صفحات فیلترشده رو (در صورت نیاز) ایندکس کنه.
// - تغییر هر فیلتر بلافاصله اعمال میشه (بدون دکمه‌ی «اعمال فیلتر»)
//   بجز محدوده‌ی قیمت که یه دکمه‌ی اعمال داره (چون تایپ عددی
//   نباید هر keystroke یه navigation جدید بسازه).
// - ردیف «فیلترهای فعال» بالای نتایج (توی خود صفحه‌ی products،
//   نه اینجا) با dismiss chip، شفافیت کامل به کاربر میده که
//   الان دقیقاً چی فیلتر شده.
// ============================================================

import { useState } from 'react';
import { useRouter, usePathname, useSearchParams } from 'next/navigation';
import { CATEGORY_GROUPS } from '@/lib/categories';
import { formatPrice } from '@/lib/format';

const PRICE_PRESETS = [
  { label: 'زیر ۵ میلیون', min: '', max: '5000000' },
  { label: '۵ تا ۲۰ میلیون', min: '5000000', max: '20000000' },
  { label: '۲۰ تا ۵۰ میلیون', min: '20000000', max: '50000000' },
  { label: 'بالای ۵۰ میلیون', min: '50000000', max: '' },
];

export default function FilterSidebar({ facets, lockedCategory, onClose }) {
  const router = useRouter();
  const pathname = usePathname();
  const searchParams = useSearchParams();

  const selectedCategories = lockedCategory ? [lockedCategory] : (searchParams.get('category')?.split(',').filter(Boolean) || []);
  const selectedBrands = searchParams.get('brand')?.split(',').filter(Boolean) || [];
  const inStockOnly = searchParams.get('inStock') === 'true';
  const isNewOnly = searchParams.get('isNew') === 'true';
  const isBestsellerOnly = searchParams.get('isBestseller') === 'true';
  const minRating = searchParams.get('minRating') || '';

  const [minPrice, setMinPrice] = useState(searchParams.get('minPrice') || '');
  const [maxPrice, setMaxPrice] = useState(searchParams.get('maxPrice') || '');

  // تابع عمومی برای آپدیت یه پارامتر توی URL (و برگردوندن page به ۱)
  function updateParam(key, value) {
    const params = new URLSearchParams(searchParams.toString());
    if (value === null || value === '' || value === false) params.delete(key);
    else params.set(key, value);
    params.delete('page');
    router.push(`${pathname}?${params.toString()}`, { scroll: false });
  }

  function toggleInList(key, value, currentList) {
    const next = currentList.includes(value)
      ? currentList.filter(v => v !== value)
      : [...currentList, value];
    updateParam(key, next.join(','));
  }

  function applyPriceRange(min = minPrice, max = maxPrice) {
    const params = new URLSearchParams(searchParams.toString());
    if (min) params.set('minPrice', min); else params.delete('minPrice');
    if (max) params.set('maxPrice', max); else params.delete('maxPrice');
    params.delete('page');
    router.push(`${pathname}?${params.toString()}`, { scroll: false });
  }

  function clearAll() {
    router.push(pathname, { scroll: false });
    setMinPrice(''); setMaxPrice('');
  }

  const hasActiveFilters = selectedCategories.length > (lockedCategory ? 1 : 0) || selectedBrands.length || inStockOnly || isNewOnly || isBestsellerOnly || minRating || minPrice || maxPrice;

  return (
    <aside className="w-full lg:w-64 shrink-0 space-y-6">
      {hasActiveFilters && (
        <button onClick={clearAll} className="text-xs text-ember hover:underline">
          ✕ پاک کردن همه‌ی فیلترها
        </button>
      )}

      {/* دسته‌بندی — فقط وقتی lockedCategory نداریم نشون بده (توی صفحه‌ی دسته، خودِ دسته مشخصه) */}

                    key={item.slug}
                    label={item.label}
                    checked={selectedCategories.includes(item.slug)}
                    onChange={() => toggleInList('category', item.slug, selectedCategories)}

              </div>

      )}

      {/* برند — از facets که سرور بر اساس محصولات موجود محاسبه کرده */}

        <FilterSection title="برند">

            {facets.brands.map(brand => (
              <CheckboxRow
                key={brand}
                label={brand}
                checked={selectedBrands.includes(brand)}
                onChange={() => toggleInList('brand', brand, selectedBrands)}

            ))}

        <div className="flex flex-wrap gap-1.5 mb-3">
          {PRICE_PRESETS.map(p => (
            <button
              key={p.label}
              onClick={() => { setMinPrice(p.min); setMaxPrice(p.max); applyPriceRange(p.min, p.max); }}
              className="text-[11px] bg-card text-ash hover:text-ink px-2.5 py-1.5 rounded-btn transition"
            >
              {p.label}
            </button>
          ))}

// ============================================================
// احراز هویت — ورود با کد یک‌بارمصرف (OTP)
// ------------------------------------------------------------
// این Context فقط از endpoint هایی استفاده می‌کنه که از قبل توی
// بک‌اند وجود دارن (همون‌هایی که پنل ادمین هم برای لاگین استفاده
// می‌کنه: /auth/send-otp ، /auth/verify-otp ، /auth/me).
// یعنی هیچ تغییری توی بک‌اند لازم نیست.
//
// 📌 درباره‌ی SMS.ir: ارسال واقعی پیامک، مسئولیت بک‌انده (داخل
// روت /auth/send-otp). اینجا فقط UI جریان ورود رو می‌سازیم. تا
// وقتی توکن SMS.ir رو وصل نکردید، هر رفتار تستی که بک‌اند فعلاً
// برای OTP داره (مثلاً چاپ کد توی کنسول یا یه کد ثابت) همینجوری
// از همین صفحه کار می‌کنه — وقتی SMS.ir رو وصل کردیم، این فرانت
// کوچیک‌ترین تغییری لازم نداره، چون فقط از endpoint موجود استفاده
// می‌کنه.
// ============================================================

import { createContext, useContext, useEffect, useState, useCallback } from 'react';
import { api, getToken, setToken } from '@/lib/api';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const token = getToken();
    if (!token) { setLoading(false); return; }

    api.getMe()
      .then(data => setUser(data.user))
      .catch(() => setToken(null)) // توکن نامعتبر/منقضی‌شده
      .finally(() => setLoading(false));
  }, []);

  const login = useCallback((token, userData) => {
    setToken(token);
    setUser(userData);
  }, []);

  const logout = useCallback(() => {
    setToken(null);
    setUser(null);
  }, []);

  return (
    <AuthContext.Provider value={{ user, loading, login, logout, isAuthenticated: !!user }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth باید داخل AuthProvider استفاده بشه');
  return ctx;
}

  useEffect(() => {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) setItems(JSON.parse(saved));
    } catch {
      // اگه داده خراب بود، سبد خالی شروع میشه
    } finally {
      setHydrated(true);
    }
  }, []);

  // ذخیره‌ی خودکار هر تغییر توی سبد
  useEffect(() => {
    if (hydrated) localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
  }, [items, hydrated]);

  const addItem = useCallback((product, quantity = 1) => {
    setItems(prev => {
      const existing = prev.find(i => i.productId === product.id);
      if (existing) {
        const newQty = Math.min(existing.quantity + quantity, product.stock ?? 99);
        return prev.map(i => i.productId === product.id ? { ...i, quantity: newQty } : i);
      }
      return [...prev, {
        productId: product.id,
        name: product.name,
        image: product.images?.[0] || '',
        price: product.price,
        stock: product.stock,
        quantity: Math.min(quantity, product.stock ?? 99),
      }];
    });
  }, []);

  const removeItem = useCallback((productId) => {
    setItems(prev => prev.filter(i => i.productId !== productId));
  }, []);

  const updateQuantity = useCallback((productId, quantity) => {
    setItems(prev => prev.map(i => {
      if (i.productId !== productId) return i;
      const clamped = Math.max(1, Math.min(quantity, i.stock ?? 99));
      return { ...i, quantity: clamped };
    }));
  }, []);

  const clearCart = useCallback(() => setItems([]), []);

  const subtotal = items.reduce((sum, i) => sum + i.price * i.quantity, 0);
  const count = items.reduce((sum, i) => sum + i.quantity, 0);

  return (
    <CartContext.Provider value={{ items, addItem, removeItem, updateQuantity, clearCart, subtotal, count, hydrated }}>
      {children}
    </CartContext.Provider>
  );
}

export function useCart() {
  const ctx = useContext(CartContext);
  if (!ctx) throw new Error('useCart باید داخل CartProvider استفاده بشه');
  return ctx;
}

// همون الگوی request() پنل ادمین، با این تفاوت که توکن از
// localStorage با کلید 'technubela_token' خونده میشه (مخصوص
// کاربر مهمان/مشتری، جدا از سشن ادمین).
//
// ⚠️ نکته‌ی خیلی مهم درباره‌ی endpoint هایی که علامت (پیشنهادی)
// دارن: من کد فعلی همه‌ی روت‌های بک‌اند رو ندیدم (فقط چندتاشون
// رو قبلاً با هم دیدیم: products, blog, auth, reviews). برای
// بقیه (فیلتر پیشرفته‌ی محصولات، سفارش‌ها، پرداخت) یه پیاده‌سازی
// پیشنهادی توی پوشه‌ی backend-additions/ گذاشتم. اگه از قبل
// نسخه‌ی دیگه‌ای داری، یا جایگزینش کن یا این فرانت رو با
// endpoint واقعی‌ت تطبیق بده.
// ============================================================

const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:5000/api';
const TOKEN_KEY = 'technubela_token';

export function getToken() {
  if (typeof window === 'undefined') return null; // سمت سرور (SSR) توکن نداریم
  return localStorage.getItem(TOKEN_KEY);
}

export function setToken(token) {
  if (typeof window === 'undefined') return;
  if (token) localStorage.setItem(TOKEN_KEY, token);
  else localStorage.removeItem(TOKEN_KEY);
}

async function request(path, options = {}) {
  const token = getToken();
  const headers = {
    ...(!(options.body instanceof FormData) && { 'Content-Type': 'application/json' }),
    ...(token && { Authorization: `Bearer ${token}` }),
    ...options.headers,
  };

  const body = options.body && !(options.body instanceof FormData) && typeof options.body !== 'string'
    ? JSON.stringify(options.body)
    : options.body;

  const res = await fetch(`${API_URL}${path}`, { ...options, headers, body });
  const data = await res.json().catch(() => ({}));

  if (!res.ok) {
    throw new Error(data.error || 'خطای سرور');
  }
  return data;
}

// ============================================================
// تبدیل searchParams صفحه به query string برای api.getProducts
// (هم توی /products و هم /category/[slug] استفاده میشه تا
// منطق فیلتر یه‌جا نگه داشته بشه)
// ============================================================
export function buildProductQuery(searchParams, { category, limit = 24 } = {}) {
  const params = new URLSearchParams();

  const categoryValue = category || searchParams.category;
  if (categoryValue) params.set('category', categoryValue);
  if (searchParams.brand) params.set('brand', searchParams.brand);
  if (searchParams.minPrice) params.set('minPrice', searchParams.minPrice);
  if (searchParams.maxPrice) params.set('maxPrice', searchParams.maxPrice);
  if (searchParams.minRating) params.set('minRating', searchParams.minRating);
  if (searchParams.inStock) params.set('inStock', searchParams.inStock);
  if (searchParams.isNew) params.set('isNew', searchParams.isNew);
  if (searchParams.isBestseller) params.set('isBestseller', searchParams.isBestseller);
  if (searchParams.search) params.set('search', searchParams.search);
  params.set('sort', searchParams.sort || 'newest');
  params.set('page', searchParams.page || '1');
  params.set('limit', String(limit));

  return `?${params.toString()}`;
}

/** قیمت رو با جداکننده‌ی هزارگان فارسی + "تومان" نمایش میده. مثلاً: ۱٬۲۵۰٬۰۰۰ تومان */
export function formatPrice(amount) {
  if (amount === null || amount === undefined || isNaN(amount)) return '—';
  return `${Number(amount).toLocaleString('fa-IR')} تومان`;
}

/** عدد ساده با اعداد فارسی (بدون واحد) — برای تعداد، امتیاز و غیره */

  if (n === null || n === undefined || isNaN(n)) return '۰';
  return Number(n).toLocaleString('fa-IR');
}

/** تاریخ کامل شمسی، مثلاً: ۱۸ مرداد ۱۴۰۵ */

}

export function discountPercent(price, oldPrice) {

// ============================================================
// ابزارهای سئو
// ------------------------------------------------------------
// نکته‌ی مهم: مدل Product توی بک‌اند فیلد slug نداره (برخلاف
// BlogPost که داره)، و طبق خواسته‌ی شما بک‌اند رو تغییر ندادیم.
// برای اینکه لینک محصولات هم برای سئو خوانا باشه (نه فقط یه
// ObjectId بی‌معنی)، از ترفند "id + اسلاگ توی یه بخش از URL"
// استفاده کردیم:
//
//   /products/64f1a2b3c4d5e6f7a8b9c0d1-rtx-4090-24gb
//
// چون ObjectId مونگو همیشه دقیقاً ۲۴ کاراکتر هگزادسیمال هست،
// می‌تونیم همیشه ۲۴ کاراکتر اول رو جدا کنیم و به بک‌اند بدیم؛
// بقیه‌ی رشته فقط برای زیبایی/سئوی URL هست و در واقع خونده
// نمیشه. این یعنی هیچ تغییری توی بک‌اند لازم نیست.
// ============================================================

const MONGO_ID_LENGTH = 24;

/** تبدیل یه متن فارسی/انگلیسی به اسلاگ URL-friendly */
export function slugify(text = '') {
  return text
    .toString()
    .trim()
    .toLowerCase()
    .replace(/\s+/g, '-')
    // حروف فارسی، انگلیسی، عدد و خط تیره رو نگه دار، بقیه حذف
    .replace(/[^\u0600-\u06FFa-z0-9-]/g, '')
    .replace(/-+/g, '-')
    .replace(/^-|-$/g, '');
}

/** ساخت URL کامل محصول (برای Link ها و og:url) */
export function productUrl(product) {
  if (!product) return '/products';
  const slug = slugify(product.name);
  return `/products/${product.id}${slug ? `-${slug}` : ''}`;
}

/** استخراج id واقعی محصول از پارامتر URL (که ممکنه id+اسلاگ باشه) */
export function extractProductId(param) {
  if (!param) return '';
  return param.slice(0, MONGO_ID_LENGTH);
}

/** JSON-LD برای صفحه‌ی محصول (schema.org/Product) — برای نمایش قیمت/موجودی/امتیاز توی نتایج گوگل */
export function buildProductJsonLd(product, siteUrl) {
  if (!product) return null;
  return {
    '@context': 'https://schema.org',
    '@type': 'Product',
    name: product.name,
    description: product.desc,
    image: product.images?.map(img => `${siteUrl}${img}`) || [],
    brand: { '@type': 'Brand', name: product.brand },
    sku: product.id,
    offers: {
      '@type': 'Offer',
      priceCurrency: 'IRR',
      price: product.price,
      availability: product.stock > 0
        ? 'https://schema.org/InStock'
        : 'https://schema.org/OutOfStock',
      url: `${siteUrl}${productUrl(product)}`,
    },
    ...(product.reviewCount > 0 && {
      aggregateRating: {
        '@type': 'AggregateRating',
        ratingValue: product.rating || 0,
        reviewCount: product.reviewCount,
      },
    }),
  };
}

/** JSON-LD برای صفحه‌ی مقاله (schema.org/Article) */
export function buildArticleJsonLd(post, siteUrl) {
  if (!post) return null;
  return {
    '@context': 'https://schema.org',
    '@type': 'Article',
    headline: post.title,
    description: post.excerpt,
    image: post.image ? [`${siteUrl}${post.image}`] : [],
    datePublished: post.date,
    dateModified: post.updatedDate || post.date,
    author: { '@type': 'Person', name: post.author?.name || 'تیم تحریریه' },
    publisher: {
      '@type': 'Organization',
      name: 'technubela',
      logo: { '@type': 'ImageObject', url: `${siteUrl}/logo.png` },
    },
    mainEntityOfPage: { '@type': 'WebPage', '@id': `${siteUrl}/blog/${post.slug}` },
  };
}

/** JSON-LD برای مسیر ناوبری (Breadcrumb) — به گوگل کمک می‌کنه ساختار سایت رو بفهمه */
export function buildBreadcrumbJsonLd(items, siteUrl) {
  return {
    '@context': 'https://schema.org',
    '@type': 'BreadcrumbList',
    itemListElement: items.map((item, i) => ({
      '@type': 'ListItem',
      position: i + 1,
      name: item.name,
      item: `${siteUrl}${item.href}`,
    })),
  };
}
